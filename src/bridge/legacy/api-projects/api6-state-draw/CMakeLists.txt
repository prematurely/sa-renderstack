cmake_minimum_required(VERSION 3.24)
project(api6_state_draw LANGUAGES CXX)
if (NOT DEFINED SA_RENDERSTACK_ROOT)
  message(FATAL_ERROR "api6-state-draw must be added through the Bridge API aggregate")
endif()
add_library(api6_state_draw_lib STATIC src/api6_state_draw.cpp)
target_compile_features(api6_state_draw_lib PUBLIC cxx_std_23)
target_include_directories(api6_state_draw_lib PUBLIC include "${SA_RENDERSTACK_SDK_INCLUDE}" "${SA_RENDERSTACK_VULKAN_INCLUDE}")
add_executable(api6_state_draw main.cpp)
target_compile_features(api6_state_draw PRIVATE cxx_std_23)
target_compile_definitions(api6_state_draw PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(api6_state_draw PRIVATE "${SA_RENDERSTACK_SDK_INCLUDE}" "${SA_RENDERSTACK_VULKAN_INCLUDE}")
target_link_libraries(api6_state_draw PRIVATE user32 api6_state_draw_lib)
target_include_directories(api6_state_draw PRIVATE include)
enable_testing()
add_executable(api6-state-draw-unit-tests tests/test_transaction.cpp)
target_link_libraries(api6-state-draw-unit-tests PRIVATE api6_state_draw_lib)
target_include_directories(api6-state-draw-unit-tests PRIVATE include)
add_test(NAME api6-state-draw-unit-tests COMMAND api6-state-draw-unit-tests)
