cmake_minimum_required(VERSION 3.24)
project(api7_selective_journal LANGUAGES CXX)
if (NOT DEFINED SA_RENDERSTACK_ROOT)
  message(FATAL_ERROR "api7-selective-journal must be added through the Bridge API aggregate")
endif()
add_library(api7_selective_journal_lib STATIC src/api7_selective_journal.cpp)
target_compile_features(api7_selective_journal_lib PUBLIC cxx_std_23)
target_include_directories(api7_selective_journal_lib PUBLIC include "${SA_RENDERSTACK_SDK_INCLUDE}" "${SA_RENDERSTACK_VULKAN_INCLUDE}")
add_executable(api7_selective_journal main.cpp)
target_compile_features(api7_selective_journal PRIVATE cxx_std_23)
target_compile_definitions(api7_selective_journal PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(api7_selective_journal PRIVATE "${SA_RENDERSTACK_SDK_INCLUDE}" "${SA_RENDERSTACK_VULKAN_INCLUDE}")
target_link_libraries(api7_selective_journal PRIVATE user32 api7_selective_journal_lib)
target_include_directories(api7_selective_journal PRIVATE include)
enable_testing()
add_executable(api7-selective-journal-unit-tests tests/test_journal.cpp)
target_link_libraries(api7-selective-journal-unit-tests PRIVATE api7_selective_journal_lib)
target_include_directories(api7-selective-journal-unit-tests PRIVATE include)
add_test(NAME api7-selective-journal-unit-tests COMMAND api7-selective-journal-unit-tests)
