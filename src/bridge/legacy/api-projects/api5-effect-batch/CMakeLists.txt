cmake_minimum_required(VERSION 3.24)
project(api5_effect_batch LANGUAGES CXX)
if (NOT DEFINED SA_RENDERSTACK_ROOT)
  message(FATAL_ERROR "api5-effect-batch must be added through the Bridge API aggregate")
endif()
add_library(api5_effect_batch_lib STATIC src/api5_effect_batch.cpp)
target_compile_features(api5_effect_batch_lib PUBLIC cxx_std_23)
target_include_directories(api5_effect_batch_lib PUBLIC include "${SA_RENDERSTACK_SDK_INCLUDE}" "${SA_RENDERSTACK_VULKAN_INCLUDE}")
add_executable(api5_effect_batch main.cpp)
target_compile_features(api5_effect_batch PRIVATE cxx_std_23)
target_compile_definitions(api5_effect_batch PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(api5_effect_batch PRIVATE api5_effect_batch_lib)
target_include_directories(api5_effect_batch PRIVATE include "${SA_RENDERSTACK_SDK_INCLUDE}" "${SA_RENDERSTACK_VULKAN_INCLUDE}")
target_link_libraries(api5_effect_batch PRIVATE user32)
enable_testing()
add_executable(api5-effect-batch-unit-tests tests/test_builder.cpp)
target_link_libraries(api5-effect-batch-unit-tests PRIVATE api5_effect_batch_lib)
add_test(NAME api5-effect-batch-unit-tests COMMAND api5-effect-batch-unit-tests)
